// Omnichannel Inbox - fill missing customer names / threads and ad details.
// Body: { limit?: number }  Called once for backfill and by pg_cron as a safety net.
import { createClient } from "npm:@supabase/supabase-js@2";

const GRAPH = "https://graph.facebook.com/v21.0";
const sb = createClient(Deno.env.get("SUPABASE_URL")!, Deno.env.get("SUPABASE_SERVICE_ROLE_KEY")!, { auth: { persistSession: false } });
const cors = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Headers": "authorization, x-client-info, apikey, content-type" };

const tokens: Record<string, string> = {};
async function token(pageId: string) {
  if (!tokens[pageId]) {
    const { data } = await sb.from("ib_page_tokens").select("token").eq("page_id", pageId).maybeSingle();
    if (data?.token) tokens[pageId] = data.token;
  }
  return tokens[pageId];
}

Deno.serve(async (req) => {
  if (req.method === "OPTIONS") return new Response("ok", { headers: cors });
  const body = await req.json().catch(() => ({}));
  const limit = Math.min(Number(body.limit) || 100, 300);
  let names = 0, pics = 0, ads = 0;

  const { data: contacts } = await sb.from("ib_contacts").select("id, platform, page_id, external_id, name, profile_pic")
    .or("name.is.null,profile_pic.is.null").is("name_checked_at", null).order("created_at", { ascending: false }).limit(limit);
  for (const c of contacts || []) {
    const t = await token(c.page_id);
    const patch: Record<string, unknown> = { name_checked_at: new Date().toISOString() };
    if (t) {
      try {
        const f = c.platform === "instagram" ? "name,username,profile_pic" : "first_name,last_name,profile_pic";
        const j = await (await fetch(`${GRAPH}/${c.external_id}?fields=${f}&access_token=${t}`)).json();
        if (!j.error) {
          const n = c.platform === "instagram" ? (j.name || j.username) : [j.first_name, j.last_name].filter(Boolean).join(" ");
          if (n && !c.name) { patch.name = n; names++; }
          if (j.profile_pic) { patch.profile_pic = j.profile_pic; pics++; }
        }
      } catch { /* ignore */ }
      if (!c.name && !patch.name) {
        try {
          const plat = c.platform === "instagram" ? "instagram" : "messenger";
          const j = await (await fetch(`${GRAPH}/${c.page_id}/conversations?platform=${plat}&user_id=${c.external_id}&fields=participants&access_token=${t}`)).json();
          const conv = j.data?.[0];
          const p = conv?.participants?.data?.find((x: any) => x.id === c.external_id) || conv?.participants?.data?.find((x: any) => x.id !== c.page_id);
          if (p?.name || p?.username) { patch.name = p.name || p.username; names++; }
          if (conv?.id) await sb.from("ib_conversations").update({ meta_thread_id: conv.id }).eq("contact_id", c.id).is("meta_thread_id", null);
        } catch { /* ignore */ }
      }
    }
    await sb.from("ib_contacts").update(patch).eq("id", c.id);
  }

  const { data: posts } = await sb.from("ib_conversations").select("source_post_id, page_id, source_ad_id").not("source_post_id", "is", null).limit(1000);
  const { data: have } = await sb.from("ib_ads").select("post_id");
  const known = new Set((have || []).map((x: any) => x.post_id));
  const seen = new Set<string>();
  for (const p of posts || []) {
    const pid = p.source_post_id as string;
    if (known.has(pid) || seen.has(pid)) continue;
    seen.add(pid);
    const row: Record<string, unknown> = { post_id: pid, page_id: p.page_id, ad_id: p.source_ad_id };
    const t = await token(p.page_id);
    if (t) {
      try {
        const j = await (await fetch(`${GRAPH}/${pid}?fields=message,full_picture,permalink_url,attachments{media_type}&access_token=${t}`)).json();
        if (!j.error) { row.message = j.message || null; row.picture = j.full_picture || null; row.permalink = j.permalink_url || null; row.media_type = j.attachments?.data?.[0]?.media_type || null; }
      } catch { /* ignore */ }
    }
    await sb.from("ib_ads").upsert(row, { onConflict: "post_id" });
    ads++;
  }
  return Response.json({ contacts: (contacts || []).length, names, pics, ads }, { headers: cors });
});
